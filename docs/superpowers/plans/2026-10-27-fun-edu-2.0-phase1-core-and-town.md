> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Dọn dẹp triệt để xung đột dependencies (GetX, Riverpod, states_rebuilder), thiết lập kiến trúc Event-Driven (`AppEventBus`, `AppBlocObserver`, `HiveStorageService`, `AudioManager`), và xây dựng màn hình trung tâm **Làng Cổ Tích Tương Tác (The Living Town)** kết nối Siêu Thị Pippo (`shop_game`), Cây Tri Thức (`wonder_tree`) và Trại Ấp Thú Cưng (`pet_sanctuary`).

**Architecture:** Mô hình quản lý trạng thái lai (Hybrid State): Dùng `flutter_bloc` với event transformers (`droppable`/`throttle`) cho Game logic chống spam touch của trẻ em, `Cubit` cho UI cục bộ, `AppEventBus` (Stream broadcast) cho liên lạc giữa các module độc lập. Toàn bộ dữ liệu lưu trữ ngoại tuyến qua `Hive` (Offline-First).

**Tech Stack:** Flutter 3.3+, Dart 3.3+, `flutter_bloc: ^8.1.1`, `get_it: ^8.0.3`, `hive_flutter: ^1.1.0`, `dio: ^5.7.0`, `just_audio: ^0.9.39`, `bloc_test: ^9.1.5`, `mocktail: ^1.0.4`.

```dart
// lib/core/event_bus/app_event_bus.dart
import 'dart:async';
import 'package:fun_edu/core/event_bus/app_events.dart';

class AppEventBus {
  static final AppEventBus _instance = AppEventBus._internal();
  factory AppEventBus() => _instance;
  AppEventBus._internal();

  final StreamController<AppEvent> _controller =
      StreamController<AppEvent>.broadcast();

  Stream<T> on<T extends AppEvent>() {
    return _controller.stream.where((event) => event is T).cast<T>();
  }

  void fire(AppEvent event) {
    if (!_controller.isClosed) {
      _controller.add(event);
    }
  }

  void dispose() {
    // Chỉ đóng khi app kết thúc
  }
}
```

```dart
// lib/app/app_bloc_observer.dart
import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:logger/logger.dart';

class AppBlocObserver extends BlocObserver {
  final void Function(Object error, StackTrace stackTrace)? onErrorLogged;
  final Logger _logger = Logger(printer: PrettyPrinter(methodCount: 0));

  AppBlocObserver({this.onErrorLogged});

  @override
  void onChange(BlocBase bloc, Change change) {
    super.onChange(bloc, change);
    _logger.d('[BLoC Change] ${bloc.runtimeType}: ${change.currentState} -> ${change.nextState}');
  }

  @override
  void onError(BlocBase bloc, Object error, StackTrace stackTrace) {
    _logger.e('[BLoC Error] in ${bloc.runtimeType}: $error');
    onErrorLogged?.call(error, stackTrace);
    super.onError(bloc, error, stackTrace);
  }
}
```

```dart
// lib/core/storage/hive_storage_service.dart
import 'package:hive_flutter/hive_flutter.dart';
import 'package:fun_edu/core/storage/storage_keys.dart';

class HiveStorageService {
  final Box _userBox;
  final Box _gameBox;

  HiveStorageService({required Box userBox, required Box gameBox})
      : _userBox = userBox,
        _gameBox = gameBox;

  int getCoins() => _userBox.get(StorageKeys.coins, defaultValue: 0) as int;

  Future<void> saveCoins(int amount) async {
    await _userBox.put(StorageKeys.coins, amount);
  }

  int getTreeWateringStreak() =>
      _gameBox.get(StorageKeys.treeWateringStreak, defaultValue: 0) as int;

  Future<void> incrementTreeWateringStreak() async {
    final current = getTreeWateringStreak();
    await _gameBox.put(StorageKeys.treeWateringStreak, current + 1);
  }

  int? getLastWateredTimestamp() =>
      _gameBox.get(StorageKeys.lastWateredTimestamp) as int?;

  Future<void> recordWateredToday() async {
    await _gameBox.put(
      StorageKeys.lastWateredTimestamp,
      DateTime.now().millisecondsSinceEpoch,
    );
  }
}
```

```dart
class AudioManager {

  final AudioPlayer _sfxPlayer = AudioPlayer();
  bool _isSoundEnabled = true;

  bool get isSoundEnabled => _isSoundEnabled;

  void toggleSound(bool enabled) {
    _isSoundEnabled = enabled;
  }

  Future<void> playSfx(AudioSfx sfx) async {
    if (!_isSoundEnabled) return;
    try {
      await _sfxPlayer.setAsset('assets/${sfx.assetPath}');

    }
  }

  void dispose() {
    _sfxPlayer.dispose();
  }
}
```

```dart
// lib/features/wonder_tree/domain/models/wonder_tree_stage.dart
enum WonderTreeStage {
  seed,       // 0 ngày
  seedling,   // 1-2 ngày
  blooming,   // 3-4 ngày
  goldenFruit // 5 ngày (Quả chín rụng rương quà)
}

extension WonderTreeStageExt on WonderTreeStage {
  static WonderTreeStage fromStreak(int streak) {
    final mod = streak % 5;
    if (mod == 0 && streak > 0) return WonderTreeStage.goldenFruit;
    if (mod <= 1) return WonderTreeStage.seed;
    if (mod <= 3) return WonderTreeStage.seedling;
    return WonderTreeStage.blooming;
  }
}
```

```dart
// lib/features/wonder_tree/presentation/bloc/wonder_tree_bloc.dart
import 'package:equatable/equatable.dart';
import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:fun_edu/core/event_bus/app_event_bus.dart';
import 'package:fun_edu/core/event_bus/app_events.dart';
import 'package:fun_edu/core/storage/hive_storage_service.dart';
import 'package:fun_edu/features/wonder_tree/domain/models/wonder_tree_stage.dart';

abstract class WonderTreeEvent extends Equatable {
  const WonderTreeEvent();
  @override
  List<Object?> get props => [];
}

class LoadWonderTreeEvent extends WonderTreeEvent {
  const LoadWonderTreeEvent();
}

class WaterTreeEvent extends WonderTreeEvent {
  const WaterTreeEvent();
}

class WonderTreeState extends Equatable {
  final int streak;
  final WonderTreeStage stage;
  final bool canWaterToday;

  const WonderTreeState({
    required this.streak,
    required this.stage,
    required this.canWaterToday,
  });

  @override
  List<Object?> get props => [streak, stage, canWaterToday];
}

class WonderTreeBloc extends Bloc<WonderTreeEvent, WonderTreeState> {
  final HiveStorageService _storage;
  final AppEventBus _eventBus;

  WonderTreeBloc({
    required HiveStorageService storage,
    required AppEventBus eventBus,
  })  : _storage = storage,
        _eventBus = eventBus,
        super(WonderTreeState(
          streak: storage.getTreeWateringStreak(),
          stage: WonderTreeStageExt.fromStreak(storage.getTreeWateringStreak()),
          canWaterToday: true,
        )) {
    on<WaterTreeEvent>(_onWaterTree);
  }

  Future<void> _onWaterTree(
    WaterTreeEvent event,
    Emitter<WonderTreeState> emit,
  ) async {
    await _storage.incrementTreeWateringStreak();
    await _storage.recordWateredToday();
    final newStreak = _storage.getTreeWateringStreak();
    final newStage = WonderTreeStageExt.fromStreak(newStreak);

    _eventBus.fire(TreeWateredEvent(currentStreak: newStreak));

    emit(WonderTreeState(
      streak: newStreak,
      stage: newStage,
      canWaterToday: false,
    ));
  }
}
```

```dart
// lib/features/pet_sanctuary/presentation/bloc/pet_hatch_bloc.dart
import 'package:equatable/equatable.dart';
import 'package:flutter_bloc/flutter_bloc.dart';

abstract class PetHatchEvent extends Equatable {
  const PetHatchEvent();
  @override
  List<Object?> get props => [];
}

class TapEggWithHammerEvent extends PetHatchEvent {
  const TapEggWithHammerEvent();
}

class PetHatchState extends Equatable {
  final int cracks;
  final bool isHatched;
  final String? petName;

  const PetHatchState({
    required this.cracks,
    required this.isHatched,
    this.petName,
  });

  @override
  List<Object?> get props => [cracks, isHatched, petName];
}

class PetHatchBloc extends Bloc<PetHatchEvent, PetHatchState> {
  PetHatchBloc({int initialCracks = 0})
      : super(PetHatchState(cracks: initialCracks, isHatched: initialCracks >= 3)) {
    on<TapEggWithHammerEvent>((event, emit) {
      final nextCracks = state.cracks + 1;
      if (nextCracks >= 3) {
        emit(PetHatchState(
          cracks: 3,
          isHatched: true,
          petName: 'Rồng Con Kẹo Bông',
        ));
      } else {
        emit(PetHatchState(
          cracks: nextCracks,
          isHatched: false,
        ));
      }
    });
  }
}
```
